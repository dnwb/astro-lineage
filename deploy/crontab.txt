# ==============================================================================
# AstroLineage arXiv 定时抓取、AI 研判与频道自动发布 Crontab 任务配置
#
# 固定时间调度策略：
#   - 阶段一（工作日固定抓取）：周一至周五 每天固定上午 10:00 (北京时间) 执行一次当日批次抓取、AI 研判与网页发布；
#     （arXiv 美东 20:00 批次于冬令时 09:00 上线，10:00 一次性稳定完成，杜绝日间重复轮询与无谓模型消耗）
#   - 阶段二（次日兜底清障对账）：周二至周六 每天凌晨 02:00 (北京时间) 执行一次前日兜底扫描与连续区间对账封板；
#   - 阶段三（周报广播）：每周一上午 08:00 (北京时间)，主动向授权群聊广播上一学术周完整周报；
#   - 统一输出日志：/tmp/arxiv-task.log
# ==============================================================================

SHELL=/bin/bash
PATH=/home/long/.npm-global/bin:/usr/local/bin:/usr/bin:/bin:/home/long/.local/bin
PROJECT_DIR=/home/long/astro-lineage-cron
CRON_TZ=Asia/Shanghai

# 阶段一：工作日固定时间抓取与研判：周一至周五上午 10:00 (北京时间)
0 10 * * 1-5 /bin/bash /home/long/astro-lineage-cron/scripts/cron-runner.sh auto >> /tmp/arxiv-task.log 2>&1

# 阶段二：次日兜底清障扫描与连续区间对账封板：周二至周六凌晨 02:00 (北京时间)
0 2 * * 2-6 /bin/bash /home/long/astro-lineage-cron/scripts/cron-runner.sh reconcile >> /tmp/arxiv-task.log 2>&1

# 阶段三：学术周报群聊定时广播：每周一上午 08:00 (北京时间)
0 8 * * 1 /bin/bash /home/long/astro-lineage-cron/scripts/cron-runner.sh group-weekly >> /tmp/arxiv-task.log 2>&1
