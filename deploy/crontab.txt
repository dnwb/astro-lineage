# ==============================================================================
# AstroLineage arXiv 定时抓取、AI 研判与频道自动发布 Crontab 任务配置
#
# 时序与执行策略：
#   - 阶段一（预运行）：周一至周五 10:00 CST，执行当日批次抓取与网页热更新，遇必读单篇即时发帖讨论；
#   - 阶段二（兜底封板）：周二至周六 02:00 CST，次日清障扫描、连续区间断言对账封板，发布整期导读总帖（周六发布周报）；
#   - 阶段三（周报广播）：每周一上午 08:00 CST，主动向授权群聊广播上一学术周完整周报；
#   - 统一输出日志：/tmp/arxiv-task.log
# ==============================================================================

SHELL=/bin/bash
PATH=/home/long/.npm-global/bin:/usr/local/bin:/usr/bin:/bin:/home/long/.local/bin
PROJECT_DIR=/home/long/axvdaily
CRON_TZ=Asia/Shanghai

# 阶段一：统一前沿学术预运行调度任务：周一至周五上午 10:00 (北京时间)
0 10 * * 1-5 /bin/bash /home/long/axvdaily/scripts/cron-runner.sh auto >> /tmp/arxiv-task.log 2>&1

# 阶段二：次日兜底清障扫描与连续区间对账封板：周二至周六凌晨 02:00 (北京时间)
0 2 * * 2-6 /bin/bash /home/long/axvdaily/scripts/cron-runner.sh reconcile >> /tmp/arxiv-task.log 2>&1

# 阶段三：学术周报群聊定时广播：每周一上午 08:00 (北京时间)
0 8 * * 1 /bin/bash /home/long/axvdaily/scripts/cron-runner.sh group-weekly >> /tmp/arxiv-task.log 2>&1
